11 /** Service interface for PostgreSQL database operations.
12   *
13   * This trait defines the contract for database interactions, providing methods
14   * for health checks and query execution. Implementations should be created via
15   * the companion object's `live` layer.
16   *
17   * All methods return ZIO effects to ensure type-safe error handling and
18   * resource management. Database operations are executed using Quill for
19   * compile-time query validation.
20   *
21   * @see
22   *   [[DatabaseService.live]] for the production implementation
23   */
24 trait DatabaseService:
25   /** Performs a basic database health check by executing SELECT 1.
26     *
27     * This method validates that the database connection is alive and can
28     * execute queries. It's used by the /health-deep endpoint to verify database
29     * availability.
30     *
31     * @return
32     *   ZIO effect that succeeds with true if the database is healthy, or fails
33     *   with an error
34     */
35   def healthCheck(): ZIO[Any, Throwable, Boolean]
36 
37 /** Companion object providing ZLayer construction.

45   /** ZLayer that provides a configured JDBC DataSource.
46     *
47     * This layer constructs a PostgreSQL DataSource with connection pooling from
48     * the provided DatabaseConfig. The DataSource is configured but not
49     * validated at this stage - validation happens in the service layer.
50     *
51     * @return
52     *   ZLayer that provides a javax.sql.DataSource instance
53     */
54   val dataSourceLayer: ZLayer[DatabaseConfig, Throwable, DataSource] =
55     ZLayer.fromZIO(
56       for {
57         config <- ZIO.service[DatabaseConfig]
58         _ <- ZIO.logInfo(
59           s"Configuring DataSource for ${config.host}:${config.port}/${config.name}"
60         )
61       } yield {
62         val ds = new org.postgresql.ds.PGSimpleDataSource()
63         ds.setServerNames(Array(config.host))
64         ds.setPortNumbers(Array(config.port))
65         ds.setDatabaseName(config.name)
66         ds.setUser(config.user)
67         ds.setPassword(config.password)
68         ds
69       }
70     )

72   /** ZLayer that provides the DatabaseService implementation.
73     *
74     * This layer requires both a Quill context and a DataSource to be provided.
75     * It performs connection validation at startup by executing a simple SELECT
76     * 1 query. If validation fails, the application will not start.
77     *
78     * The returned service implementation provides:
79     *   - Health check via SELECT 1
80     *   - Raw SQL execution (for future use)
81     *   - Query methods (not yet implemented)
82     *
83     * @return
84     *   ZLayer that provides DatabaseService, requiring Quill and DataSource
85     */
86   private val serviceLayer: ZLayer[
87     Quill.Postgres[SnakeCase.type] & DataSource,
88     Throwable,
89     DatabaseService
90   ] =
91     ZLayer.fromZIO(
92       for {
93         ctx <- ZIO.service[Quill.Postgres[SnakeCase.type]]
94         ds <- ZIO.service[DataSource]
95 
96         // Validate connection at startup
97         _ <- ZIO
98           .attemptBlocking {
99             val conn = ds.getConnection()
100             try {
101               val stmt = conn.createStatement()
102               val rs = stmt.executeQuery("SELECT 1")
103               if (rs.next() && rs.getInt(1) == 1) {
104                 Right(())
105               } else {
106                 Left(
107                   new RuntimeException(
108                     "Database connection validation failed: unexpected result"
109                   )
110                 )
111               }
112             } finally {
113               conn.close()
114             }
115           }
116           .flatMap(ZIO.fromEither)
117         _ <- ZIO.logInfo("Database connection validated successfully")
118 
119         service = new DatabaseService {
120           import ctx._
121 
122           override def healthCheck(): ZIO[Any, Throwable, Boolean] =
123             ctx
124               .run(sql"SELECT 1".as[Query[Int]])
125               .map { results =>
126                 results.headOption.contains(1)
127               }
128         }
129       } yield service
130     )

132   /** Complete ZLayer stack that provides a production-ready DatabaseService.
133     *
134     * This layer combines:
135     *   - DatabaseConfig from environment variables
136     *   - DataSource configuration
137     *   - Quill PostgreSQL context with SnakeCase naming strategy
138     *   - Service implementation with startup validation
139     *
140     * This is the primary layer to use when providing DatabaseService to your
141     * application. It handles all dependency wiring and ensures the database is
142     * accessible before the application starts serving requests.
143     *
144     * @return
145     *   ZLayer that provides DatabaseService
146     */
147   val live: ZLayer[Any, Throwable, DatabaseService] =
148     ZLayer.make[DatabaseService](
149       DatabaseConfig.layer.mapError(e => new RuntimeException(e.getMessage)),
150       dataSourceLayer,
151       Quill.Postgres.fromNamingStrategy(SnakeCase),
152       serviceLayer
153     )

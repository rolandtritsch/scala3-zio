133 object AppConfig:
134   /** ZLayer that loads complete AppConfig from environment variables.
135     *
136     * This layer combines all three sub-configurations and will fail fast at
137     * application startup if any required configuration is missing.
138     */
139   val layer: ZLayer[Any, Throwable, AppConfig] =
140     ZLayer.make[AppConfig](
141       ServerConfig.layer.mapError(e => new RuntimeException(e.getMessage)),
142       DatabaseConfig.layer.mapError(e => new RuntimeException(e.getMessage)),
143       AwsConfig.layer.mapError(e => new RuntimeException(e.getMessage)),
144       ZLayer.fromZIO(
145         for {
146           server <- ZIO.service[ServerConfig]
147           database <- ZIO.service[DatabaseConfig]
148           aws <- ZIO.service[AwsConfig]
149         } yield AppConfig(server, database, aws)
150       )
151     )

6 /** Base class for HTTP endpoint implementations.
7   *
8   * This abstract class provides common functionality for all HTTP endpoints,
9   * including automatic request logging with timing information. Concrete
10   * endpoint implementations should extend this class and define their route and
11   * handler.
12   *
13   * The `withLogging` method can be used to wrap any handler with automatic
14   * logging of request start, completion, duration, and HTTP status code.
15   *
16   * @see
17   *   [[withLogging]] for request logging functionality
18   */
19 abstract class Endpoint:
20   val route: Route[Any, Nothing]
21   protected val handler: Handler[Any, Nothing, Request, Response]
22 
23   /** Wraps a handler with request logging (start, end, duration, status)
24     *
25     * @param handler
26     *   The handler to wrap with logging
27     * @return
28     *   A new handler that logs request start/end with timing and status
29     */
30   protected def withLogging(
31       handler: Handler[Any, Nothing, Request, Response]
32   ): Handler[Any, Nothing, Request, Response] =
33     Handler.fromFunctionZIO { (request: Request) =>
34       val method = request.method.toString
35       val path = request.path.encode
36 
37       ZIO.scoped {
38         for {
39           _ <- ZIO.logInfo(s"$method $path - Request started")
40           startTime <- Clock.nanoTime
41           response <- handler(request)
42           endTime <- Clock.nanoTime
43           durationMs = (endTime - startTime) / 1_000_000
44           status = response.status.code
45           _ <- ZIO.logInfo(
46             s"$method $path - Request completed (${durationMs}ms, status: $status)"
47           )
48         } yield response
49       }
50     }

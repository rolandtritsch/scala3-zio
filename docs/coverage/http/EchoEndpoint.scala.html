5 /** HTTP endpoint that echoes back the request body.
6   *
7   * This endpoint accepts POST requests at `/echo` and returns the request body
8   * as plain text in the response. It's useful for testing and debugging HTTP
9   * clients.
10   *
11   * '''Example usage:'''
12   * {{{
13   *   curl -X POST http://localhost:8080/echo -d "Hello, World!"
14   *   # Response: Hello, World!
15   * }}}
16   *
17   * The endpoint includes automatic request logging via the `withLogging`
18   * wrapper, which logs request start, completion time, and HTTP status.
19   */
20 object EchoEndpoint extends Endpoint:
21   override protected final val handler = withLogging(
22     Handler.fromFunctionZIO { (request: Request) =>
23       request.body.asString.orDie.map(Response.text(_))
24     }
25   )
26   override val route = Method.POST / "echo" -> handler

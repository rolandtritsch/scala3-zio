6 /** HTTP endpoint that triggers graceful server shutdown.
7   *
8   * This endpoint accepts POST requests at `/shutdown` and initiates a graceful
9   * shutdown sequence. When called:
10   *   1. Immediately returns HTTP 202 Accepted 2. Signals the server to begin
11   *      shutdown via the provided Promise 3. Allows in-flight requests to
12   *      complete (30-second timeout) 4. Cleanly terminates the application
13   *
14   * '''Endpoint:''' POST /shutdown
15   *
16   * '''Response:''' HTTP 202 Accepted with "Shutdown initiated" message
17   *
18   * '''Security Note:''' This endpoint has no authentication. In production, you
19   * should protect it with authentication/authorization or restrict access via
20   * network policies.
21   *
22   * @see
23   *   [[Main.performGracefulShutdown]] for the shutdown implementation
24   */
25 object ShutdownEndpoint:
26 
27   /** Creates a shutdown route that triggers graceful server shutdown
28     *
29     * @param shutdownPromise
30     *   Promise to complete when shutdown is requested
31     * @return
32     *   Route that handles POST /shutdown requests
33     */
34   def route(shutdownPromise: Promise[Nothing, Unit]): Route[Any, Nothing] =
35     val baseHandler = Handler.fromZIO {
36       ZIO
37         .logInfo("Shutdown endpoint called, initiating graceful shutdown...") *>
38         shutdownPromise.succeed(()).forkDaemon *>
39         ZIO.succeed(
40           Response
41             .status(Status.Accepted)
42             .addHeader(Header.ContentType(MediaType.text.plain))
43             .copy(body = Body.fromString("Shutdown initiated"))
44         )
45     }
46 
47     val handler = Handler.fromFunctionZIO { (request: Request) =>
48       val method = request.method.toString
49       val path = request.path.encode
50 
51       ZIO.scoped {
52         for {
53           _ <- ZIO.logInfo(s"$method $path - Request started")
54           startTime <- Clock.nanoTime
55           response <- baseHandler(request)
56           endTime <- Clock.nanoTime
57           durationMs = (endTime - startTime) / 1_000_000
58           status = response.status.code
59           _ <- ZIO.logInfo(
60             s"$method $path - Request completed (${durationMs}ms, status: $status)"
61           )
62         } yield response
63       }
64     }

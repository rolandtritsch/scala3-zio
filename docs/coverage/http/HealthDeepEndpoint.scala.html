20 /** Comprehensive health check endpoint that validates all external
21   * dependencies.
22   *
23   * This endpoint performs deep health checks on:
24   *   - External URL availability (https://tedn.life)
25   *   - AWS S3 connectivity and credentials
26   *   - PostgreSQL database connectivity
27   *
28   * Each check runs with a timeout and returns a detailed status. The endpoint
29   * returns HTTP 200 only if all checks pass, otherwise HTTP 500.
30   *
31   * '''Endpoint:''' GET /health-deep
32   *
33   * '''Response Format:'''
34   * {{{
35   * {
36   *   "url": {"status": "healthy", "message": "URL returned status 200"},
37   *   "s3": {"status": "healthy", "message": "Successfully listed 5 bucket(s)"},
38   *   "database": {"status": "healthy", "message": "Database connection successful"}
39   * }
40   * }}}
41   *
42   * '''Configuration Required:'''
43   *   - `AWS_ACCESS_KEY_ID` - AWS access key (optional but required for S3
44   *     check)
45   *   - `AWS_SECRET_ACCESS_KEY` - AWS secret key (optional but required for S3
46   *     check)
47   *   - `AWS_REGION` - AWS region (default: us-east-1)
48   *   - Database credentials (always required, see
49   *     [[org.roland.scala3_zio_template.service.DatabaseService]])
50   *
51   * If AWS credentials are missing, the endpoint returns HTTP 500 with error
52   * details but the application continues running.
53   *
54   * @see
55   *   [[HealthEndpoint]] for basic liveness checks without external dependencies
56   */

128   /** Performs an HTTP health check on the specified URL.
129     *
130     * The check succeeds if the URL returns a successful HTTP status (2xx).
131     * Includes a 5-second timeout to prevent hanging on unresponsive servers.
132     *
133     * @param url
134     *   The URL to check (e.g., "https://tedn.life")
135     * @return
136     *   ZIO effect that always succeeds with a ServiceHealthCheck result
137     */
138   private def checkUrl(
139       url: String
140   ): ZIO[Client & Scope, Nothing, ServiceHealthCheck] =
141     Client
142       .batched(Request.get(url))
143       .timeout(5.seconds)
144       .map {
145         case Some(resp) if resp.status.isSuccess =>
146           ServiceHealthCheck(
147             status = "healthy",
148             message = s"URL returned status ${resp.status.code}"
149           )
150         case Some(resp) =>
151           ServiceHealthCheck(
152             status = "unhealthy",
153             message = s"URL returned non-success status: ${resp.status.code}"
154           )
155         case None =>
156           ServiceHealthCheck(
157             status = "unhealthy",
158             message = "URL request timed out after 5 seconds"
159           )
160       }
161       .catchAll { error =>
162         ZIO.succeed(
163           ServiceHealthCheck(
164             status = "unhealthy",
165             message = s"URL check failed: ${error.getMessage}"
166           )
167         )
168       }

170   /** Performs a health check on AWS S3 by listing buckets.
171     *
172     * The check succeeds if the S3 client can authenticate and list buckets.
173     * Includes a 10-second timeout to prevent hanging on network issues.
174     *
175     * @param s3Layer
176     *   ZLayer providing the configured S3 client
177     * @return
178     *   ZIO effect that always succeeds with a ServiceHealthCheck result
179     */
180   private def checkS3(
181       s3Layer: ZLayer[Any, Throwable, S3]
182   ): ZIO[Any, Nothing, ServiceHealthCheck] =
183     (for {
184       buckets <- S3.listBuckets(ListBucketsRequest()).runCollect
185       bucketCount = buckets.size
186     } yield ServiceHealthCheck(
187       status = "healthy",
188       message = s"Successfully listed $bucketCount bucket(s)"
189     )).timeout(10.seconds)
190       .map {
191         case Some(result) => result
192         case None =>
193           ServiceHealthCheck(
194             status = "unhealthy",
195             message = "S3 list-buckets operation timed out after 10 seconds"
196           )
197       }
198       .provideLayer(s3Layer)
199       .catchAll { error =>
200         val errorMsg = error match
201           case e: AwsError  => e.toString
202           case t: Throwable => t.getMessage
203         ZIO.succeed(
204           ServiceHealthCheck(
205             status = "unhealthy",
206             message = s"S3 check failed: $errorMsg"
207           )
208         )
209       }

245   /** HTTP handler that performs all health checks and returns a combined
246     * response.
247     *
248     * This handler:
249     *   1. Gets AWS configuration from the environment 2. Runs URL, S3, and
250     *      database checks in sequence 3. Combines results into a JSON response
251     *      4. Returns HTTP 200 if all checks pass, HTTP 500 if any fail
252     */
253   protected final val handler
254       : Handler[DatabaseService & AwsConfig, Nothing, Request, Response] =
255     Handler.fromFunctionZIO[Request] { _ =>
256       ZIO
257         .service[AwsConfig]
258         .flatMap { awsConfig =>
259           val s3Layer = createS3Layer(awsConfig)
260 
261           // Run all three checks
262           val urlCheck: ZIO[Any, Nothing, ServiceHealthCheck] =
263             checkUrl("https://tedn.life")
264               .provideLayer((Client.default ++ Scope.default).orDie)
265           val s3Check: ZIO[Any, Nothing, ServiceHealthCheck] =
266             checkS3(s3Layer)
267 
268           for {
269             urlResult <- urlCheck
270             s3Result <- s3Check
271             dbResult <- checkDatabase
272 
273             response = HealthCheckResponse(
274               url = urlResult,
275               s3 = s3Result,
276               database = dbResult
277             )
278             isHealthy = urlResult.status == "healthy" &&
279               s3Result.status == "healthy" &&
280               dbResult.status == "healthy"
281             statusCode =
282               if isHealthy then Status.Ok else Status.InternalServerError
283           } yield Response.json(response.toJson).copy(status = statusCode)
284         }
285     }

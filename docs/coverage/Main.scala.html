87   /** Performs graceful shutdown sequence with logging.
88     *
89     * This method handles the shutdown process by:
90     *   1. Logging the shutdown initiation 2. Waiting briefly (1 second) for
91     *      in-flight requests to complete 3. Logging successful completion 4.
92     *      Preparing for application exit
93     *
94     * The 1-second delay allows the HTTP server to finish processing any
95     * requests that were in flight when shutdown was triggered.
96     *
97     * @return
98     *   ZIO effect that performs shutdown sequence and never fails
99     */
100   private def performGracefulShutdown: UIO[Unit] =
101     ZIO.logInfo("Shutdown signal received, stopping server...") *>
102       ZIO.logInfo(
103         "Waiting for in-flight requests to complete (30s timeout)..."
104       ) *>
105       ZIO.sleep(1.second) *>
106       ZIO.logInfo("Server shutdown completed successfully") *>
107       ZIO.logInfo("Exiting application...")

109   /** Main application logic that starts the HTTP server and handles shutdown.
110     *
111     * This method orchestrates the entire application lifecycle:
112     *   1. Loads server configuration from environment variables 2. Creates a
113     *      shutdown promise for coordination 3. Configures all HTTP routes 4.
114     *      Starts the HTTP server on configured port 5. Races server execution
115     *      against shutdown signal 6. Handles interrupt signals (Ctrl+C) with
116     *      graceful shutdown 7. Provides all required dependencies (Server,
117     *      DatabaseService, AwsConfig)
118     *
119     * The server will continue running until either:
120     *   - A POST request is made to /shutdown endpoint
121     *   - An interrupt signal (Ctrl+C) is received
122     *
123     * Both cases trigger graceful shutdown with a 30-second timeout.
124     *
125     * The application will fail fast at startup if any required configuration is
126     * missing (DATABASE_USER, DATABASE_PASSWORD, AWS_ACCESS_KEY_ID,
127     * AWS_SECRET_ACCESS_KEY).
128     *
129     * @return
130     *   ZIO effect that runs the complete application
131     */
132   def run: ZIO[ZIOAppArgs & Scope, Any, Any] =
133     (for {
134       serverConfig <- ZIO.service[ServerConfig]
135       _ <- ZIO.logInfo(s"Starting server on port ${serverConfig.port}...")
136       shutdownPromise <- Promise.make[Nothing, Unit]
137       allRoutes = routes(shutdownPromise)
138       server <- Server
139         .serve(allRoutes)
140         .race(shutdownPromise.await *> performGracefulShutdown)
141         .onInterrupt(
142           ZIO.logInfo(
143             "Interrupt received, shutting down..."
144           ) *> performGracefulShutdown
145         )
146         .fork
147       _ <- server.join
148     } yield ()).provide(

151       Server.live,

154     )
